```python
# Program to calculate the first 10,000 terms of the series 
# 1 - 1/3 + 1/5 - 1/7 + ... and multiply the total by 4

def calculate_series(terms):
    total = 0
    for n in range(terms):
        term = (-1)**n / (2*n + 1)  # calculate the nth term
        total += term  # accumulate the value of the series

    result = total * 4  # multiply the total by 4
    return result

# Calculate the first 10,000 terms
num_terms = 10000
series_result = calculate_series(num_terms)

# Print the result
print("The result of the series after multiplying by 4 is:", series_result)
```

### Output
```text
The result of the series after multiplying by 4 is: 3.1415926535897733
```
